# Amber CLI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A terminal client (`amber cli`) that renders assistant markdown (including tables) as ANSI terminal codes over an append-only, batched output stream sized for a 6 kbps SSH link.

**Architecture:** New `src/cli/` modules driven by the existing HTTP API: one SSE subscription (`GET /api/sessions/:id/events`) is the single event source (the server mirrors run events to subscribers), `POST /messages` etc. are fire-and-drain. Pure modules (renderer, block splitter, tool summaries, event-display reducer — `display.ts`, split out of `main.ts` for size) are unit-tested with golden strings; the REPL wires them together.

**Tech Stack:** TypeScript (strict, NodeNext, ESM `.js` imports), Node ≥18.17 built-ins only (`node:readline/promises`, global `fetch`), existing `markdown-it` dependency (ships its own types).

**Spec:** `docs/superpowers/specs/2026-10-24-amber-cli-design.md` — read it before starting; this plan argues from it.

## Global Constraints

- No new runtime or dev dependencies.
- Never print: tool/bash output, file contents, thinking text (status line mentions thinking char counts only).
- Output is append-only; the only in-place rewrite is one status line via `\r\x1b[K`.
- Every byte prints exactly once; no re-rendering of already-printed content.
- TypeScript strict + `noUncheckedIndexedAccess` + `exactOptionalPropertyTypes`; ESM imports end in `.js`.
- Comments sparse; tests use `node:test` + `node:assert/strict` with plain data fixtures; test files import from `../src/....js`.
- Run `npm test` before finishing any task. Commit messages: one imperative line.
- Wire protocol facts (verified in this repo): run events go to BOTH the `POST /messages` response and `/events` subscribers (`emitSessionEvent` in `server-app.ts`); `store.list()` returns sessions sorted by `updatedAt` descending.

```ts
/** Append-only terminal writer. All output batches until flushed; the only
 * in-place rewrite is a single status line, so a slow link carries few, full
 * packets and never a repaint. */
export class Terminal {
  private buffer = "";
  private timer: NodeJS.Timeout | null = null;
  private statusText: string | null = null;
  readonly color: boolean;

  constructor(
    private readonly out: Pick<NodeJS.WriteStream, "write" | "isTTY" | "columns"> =
      process.stdout as NodeJS.WriteStream,
    private readonly flushIntervalMs = 250,
    private readonly maxBufferBytes = 4096,
  ) {
    this.color = this.out.isTTY === true && process.env.NO_COLOR === undefined;
  }

  get interactive(): boolean { return this.out.isTTY === true; }
  get columns(): number {
    const columns = this.out.columns;
    return columns && columns > 0 ? columns : 80;
  }

  style(text: string, codes: string): string {
    return this.color ? `\x1b[${codes}m${text}\x1b[0m` : text;
  }

  append(text: string): void {
    this.buffer += text;
    if (Buffer.byteLength(this.buffer, "utf8") >= this.maxBufferBytes) {
      this.flush();
      return;
    }
    if (this.timer === null) {
      this.timer = setTimeout(() => this.flush(), this.flushIntervalMs);
      this.timer.unref?.();
    }
  }

  line(text = ""): void { this.append(`${text}\n`); }

  status(text: string | null): void {
    if (!this.interactive || this.statusText === text) return;
    this.flush(); // pending content must land above the status line
    this.clearStatus();
    this.statusText = text;
    this.paintStatus();
  }

  suppressStatus(): void {
    if (this.statusText === null) return;
    this.clearStatus();
    this.statusText = null;
  }

  flush(): void {
    if (this.timer !== null) {
      clearTimeout(this.timer);
      this.timer = null;
    }
    if (this.buffer === "") return;
    this.clearStatus();
    this.out.write(this.buffer);
    this.buffer = "";
    this.paintStatus();
  }

  end(): void {
    this.status(null);
    this.flush();
  }

  private clearStatus(): void {
    if (this.statusText !== null) this.out.write("\r\x1b[K");
  }

  private paintStatus(): void {
    if (this.statusText !== null) this.out.write(`\r\x1b[K${this.statusText}`);
  }
}
```

```ts
// src/cli/render.ts
import MarkdownIt from "markdown-it";

export interface RenderOptions { width: number; color: boolean; }

export const BOLD = "1";
export const DIM = "2";
export const ITALIC = "3";
export const CYAN = "36";
export const RED = "31";

type Run = { text: string; codes?: string };
type Line = Run[];
type Token = ReturnType<MarkdownIt["parse"]>[number];

const md = new MarkdownIt({ html: false, linkify: false, typographer: false, breaks: false });

export function displayWidth(text: string): number {
  let width = 0;
  for (const character of text) {
    const code = character.codePointAt(0)!;
    if (isZeroWidth(code)) continue;
    width += isWide(code) ? 2 : 1;
  }
  return width;
}

function isZeroWidth(code: number): boolean {
  return (code >= 0x0300 && code <= 0x036f) || (code >= 0x200b && code <= 0x200f)
    || (code >= 0xfe00 && code <= 0xfe0f);
}

function isWide(code: number): boolean {
  return (code >= 0x1100 && code <= 0x115f) || (code >= 0x2e80 && code <= 0x303e)
    || (code >= 0x3041 && code <= 0x33ff) || (code >= 0x3400 && code <= 0x4dbf)
    || (code >= 0x4e00 && code <= 0x9fff) || (code >= 0xa000 && code <= 0xa4cf)
    || (code >= 0xac00 && code <= 0xd7a3) || (code >= 0xf900 && code <= 0xfaff)
    || (code >= 0xfe10 && code <= 0xfe19) || (code >= 0xfe30 && code <= 0xfe6f)
    || (code >= 0xff00 && code <= 0xff60) || (code >= 0xffe0 && code <= 0xffe6)
    || (code >= 0x1f300 && code <= 0x1f64f) || (code >= 0x1f900 && code <= 0x1f9ff)
    || (code >= 0x20000 && code <= 0x3fffd);
}

function joinCodes(left: string, right: string): string {
  return [left, right].filter((part) => part !== "").join(";");
}

/** Greedy word wrap over styled runs; long words hard-split at the width. */
function wrapRuns(runs: Run[], width: number): Line[] {
  if (width < 1) return [runs];
  const lines: Line[] = [[]];
  let used = 0;
  const newLine = () => { lines.push([]); used = 0; };
  for (const run of runs) {
    for (const piece of run.text.split(/(\s+)/).filter((part) => part !== "")) {
      if (/^\s+$/.test(piece)) {
        const line = lines[lines.length - 1]!;
        if (line.length === 0 || used + 1 > width) continue;
        const last = line[line.length - 1]!;
        if (last.codes === run.codes) last.text += " ";
        else line.push({ text: " ", ...(run.codes ? { codes: run.codes } : {}) });
        used += 1;
      } else {
        let word = piece;
        while (word !== "" && used > 0 && displayWidth(word) > width - used) newLine();
        while (word !== "" && displayWidth(word) > width) {
          let take = 0;
          let takeWidth = 0;
          for (const character of word) {
            const characterWidth = displayWidth(character);
            if (takeWidth + characterWidth > width) break;
            take += character.length;
            takeWidth += characterWidth;
          }
          lines[lines.length - 1]!.push({ text: word.slice(0, take), ...(run.codes ? { codes: run.codes } : {}) });
          word = word.slice(take);
          newLine();
        }
        if (word !== "") {
          lines[lines.length - 1]!.push({ text: word, ...(run.codes ? { codes: run.codes } : {}) });
          used += displayWidth(word);
        }
      }
    }
  }
  return lines;
}

/** Inline tokens to styled lines (hardbreaks split lines, softbreaks join). */
function renderInline(children: Token[] | null | undefined): Line[] {
  const lines: Line[] = [[]];
  const styles: string[] = [];
  const link = { href: "", text: "" };
  const current = () => styles.join(";");
  const push = (text: string, codes: string) => {
    if (!text) return;
    lines[lines.length - 1]!.push({ text, ...(codes ? { codes } : {}) });
    if (link.href !== "") link.text += text;
  };
  for (const token of children ?? []) {
    switch (token.type) {
      case "text": push(token.content, current()); break;
      case "code_inline": push(token.content, joinCodes(current(), CYAN)); break;
      case "softbreak": push(" ", current()); break;
      case "hardbreak": lines.push([]); break;
      case "strong_open": styles.push(BOLD); break;
      case "strong_close": styles.pop(); break;
      case "em_open": styles.push(ITALIC); break;
      case "em_close": styles.pop(); break;
      case "s_open": case "s_close": break;
      case "link_open":
        link.href = token.attrGet("href") ?? "";
        link.text = "";
        break;
      case "link_close": {
        const href = link.href;
        link.href = "";
        if (href !== "" && href !== link.text) push(` (${href})`, DIM);
        break;
      }
      case "image": {
        const alt = (token.children ?? []).map((child) => child.content).join("");
        push(`[img: ${alt}]`, DIM);
        break;
      }
      default: break; // html_inline and anything exotic render as nothing
    }
  }
  return lines;
}

class BlockRenderer {
  private readonly lines: Line[] = [];
  private blankBefore = false;
  private readonly prefixes: Array<{ first: Run[]; rest: Run[] }> = [];

  constructor(private readonly options: RenderOptions) {}

  private blank(): void {
    if (this.prefixes.length === 0) this.blankBefore = true;
  }

  private restPrefix(): Run[] {
    return this.prefixes.flatMap((prefix) => prefix.rest);
  }

  private widthFor(): number {
    return Math.max(1, this.options.width - displayWidth(this.restPrefix().map((run) => run.text).join("")));
  }

  private emit(wrapped: Line[]): void {
    if (this.blankBefore && this.lines.length > 0) this.lines.push([]);
    this.blankBefore = false;
    const first = this.prefixes.flatMap((prefix) => prefix.first);
    const rest = this.restPrefix();
    if (wrapped.length === 0) wrapped = [[]];
    wrapped.forEach((line, index) => {
      this.lines.push([...(index === 0 ? first : rest), ...line]);
    });
  }

  private emitCode(body: string): void {
    if (this.blankBefore && this.lines.length > 0) this.lines.push([]);
    this.blankBefore = false;
    const prefix = [...this.restPrefix(), { text: "  ", codes: DIM }];
    for (const codeLine of body.replace(/\n$/, "").split("\n")) {
      this.lines.push([...prefix, { text: codeLine }]);
    }
  }

  render(tokens: Token[]): void {
    for (let i = 0; i < tokens.length; i += 1) {
      const token = tokens[i]!;
      switch (token.type) {
        case "heading_open": {
          this.blank();
          const runs = renderInline(tokens[i + 1]?.children).flat().map((run) => ({ ...run, codes: BOLD }));
          this.emit([runs]);
          i += 2;
          break;
        }
        case "paragraph_open": {
          this.blank();
          const wrapped = renderInline(tokens[i + 1]?.children).flatMap((line) => wrapRuns(line, this.widthFor()));
          this.emit(wrapped);
          i += 2;
          break;
        }
        case "fence":
        case "code_block":
          this.blank();
          this.emitCode(token.content);
          break;
        case "hr":
          this.blank();
          this.emit([[{ text: "--------", codes: DIM }]]);
          break;
        default: break; // lists, quotes, tables arrive in Task 3
      }
    }
  }

  finish(): string {
    while (this.lines.length > 0 && this.lines[this.lines.length - 1]!.length === 0) this.lines.pop();
    let start = 0;
    while (start < this.lines.length && this.lines[start]!.length === 0) start += 1;
    return this.lines.slice(start).map((line) => this.toText(line)).join("\n");
  }

  private toText(line: Line): string {
    if (!this.options.color) return line.map((run) => run.text).join("");
    let out = "";
    let active = "";
    for (const run of line) {
      const codes = run.codes ?? "";
      if (codes !== active) {
        if (active !== "") out += "\x1b[0m";
        if (codes !== "") out += `\x1b[${codes}m`;
        active = codes;
      }
      out += run.text;
    }
    if (active !== "") out += "\x1b[0m";
    return out;
  }
}

export function renderMarkdown(markdown: string, options: RenderOptions): string {
  const renderer = new BlockRenderer(options);
  renderer.render(md.parse(markdown, {}));
  return renderer.finish();
}
```

```ts
    for (let i = start + 1; i < tokens.length; i += 1) {
      const token = tokens[i]!;
      if (token.type === "table_close") { close = i; break; }
      if (token.type === "thead_open") header = true;
      else if (token.type === "thead_close") header = false;
      else if (token.type === "tr_open") rows.push({ header, cells: [], align: [] });
      else if (token.type === "th_open" || token.type === "td_open") {
        const row = rows[rows.length - 1]!;
        row.cells.push(renderInline(tokens[i + 1]?.children).flat());
        const style = token.attrGet?.("style") ?? "";
        row.align.push(style.includes("right") ? "right" : style.includes("center") ? "center" : "left");
      }
    }
    const columnCount = Math.max(1, ...rows.map((row) => row.cells.length));
    const plain = (line: Line) => line.map((run) => run.text).join("");
    const widths: number[] = [];
    for (let column = 0; column < columnCount; column += 1) {
      widths[column] = 1;
      for (const row of rows) {
        widths[column] = Math.max(widths[column]!, displayWidth(plain(row.cells[column] ?? [])));
      }
    }
    const budget = this.widthFor() - (columnCount * 3 + 1);
    while (widths.reduce((sum, width) => sum + width, 0) > budget && Math.max(...widths) > 1) {
      const widest = widths.indexOf(Math.max(...widths));
      widths[widest]! -= 1;
    }
    const pipe: Run = { text: "|", codes: DIM };
    const pad = (line: Line, width: number, align: string, codes?: string): Line => {
      const runs = codes === undefined ? line : line.map((run) => ({ ...run, codes: joinCodes(run.codes ?? "", codes) }));
      const gap = width - displayWidth(plain(runs));
      const spaces = (count: number): Run[] => (count > 0 ? [{ text: " ".repeat(count) }] : []);
      if (align === "right") return [...spaces(gap), ...runs];
      if (align === "center") return [...spaces(Math.floor(gap / 2)), ...runs, ...spaces(gap - Math.floor(gap / 2))];
      return [...runs, ...spaces(gap)];
    };
    const table: Line[] = [];
    for (const row of rows) {
      const cellLines = row.cells.map((cell, column) =>
        wrapRuns(cell, widths[column]!).map((line) => pad(line, widths[column]!, row.align[column] ?? "left")));
      const height = Math.max(1, ...cellLines.map((lines) => lines.length));
      const bold = row.header ? BOLD : undefined;
      for (let visual = 0; visual < height; visual += 1) {
        const line: Line = [pipe];
        for (let column = 0; column < columnCount; column += 1) {
          const cell = cellLines[column]?.[visual] ?? pad([], widths[column]!, row.align[column] ?? "left");
          line.push(...(bold === undefined ? cell : cell.map((run) => ({ ...run, codes: bold }))), pipe);
        }
        table.push(line);
      }
      if (row.header) {
        table.push(widths.flatMap((width) => [pipe, { text: "-".repeat(width + 2), codes: DIM }]).slice(0, -1).concat(pipe));
      }
    }
    this.blank();
    this.emit(table);
    return close;
  }
```

Note: the header row pads each cell to its column width, so `pad` must run on header cells too — the `bold` styling is applied over already-padded runs, which is what the golden test encodes.

```ts
export interface BlockSplitter {
  /** Feeds a delta; returns blocks closed by it. */
  push(delta: string): string[];
  /** Returns and clears the pending remainder. */
  finish(): string;
}

interface FenceState { marker: string; length: number }

export function createBlockSplitter(): BlockSplitter {
  let buffer = "";
  let pending: string[] = [];
  let fenceLines: string[] = [];
  let fence: FenceState | null = null;
  const closed: string[] = [];

  const fenceOf = (line: string): FenceState | null => {
    const match = line.match(/^ {0,3}(`{3,}|~{3,})/);
    return match ? { marker: match[1]![0]!, length: match[1]!.length } : null;
  };
  const closePending = () => {
    while (pending.length > 0 && pending[0] === "") pending = pending.slice(1);
    if (pending.length > 0) closed.push(pending.join("\n"));
    pending = [];
  };

  return {
    push(delta: string): string[] {
      closed.length = 0;
      buffer += delta;
      for (;;) {
        const newline = buffer.indexOf("\n");
        if (newline === -1) break;
        const line = buffer.slice(0, newline);
        buffer = buffer.slice(newline + 1);
        if (fence) {
          fenceLines.push(line);
          const closing = fenceOf(line);
          if (closing && closing.marker === fence.marker && closing.length >= fence.length) {
            closed.push(fenceLines.join("\n"));
            fenceLines = [];
            fence = null;
          }
          continue;
        }
        const opening = fenceOf(line);
        if (opening) {
          closePending();
          fence = opening;
          fenceLines = [line];
        } else if (line.trim() === "") {
          closePending();
        } else {
          pending.push(line);
        }
      }
      return [...closed];
    },
    finish(): string {
      const parts: string[] = [];
      if (fence) {
        parts.push(...fenceLines, buffer);
        fenceLines = [];
        fence = null;
        buffer = "";
      } else if (buffer !== "") {
        pending.push(buffer);
        buffer = "";
      }
      closePending();
      parts.push(...closed);
      closed.length = 0;
      return parts.join("\n");
    },
  };
}
```

```ts
import { homedir } from "node:os";
import type { ToolCall } from "../types.js";

function collapse(text: string): string {
  return text.replace(/\s+/g, " ").trim();
}

function truncate(text: string, max: number): string {
  const flat = collapse(text);
  return flat.length > max ? `${flat.slice(0, max - 1)}…` : flat;
}

function displayPath(path: string): string {
  const home = homedir();
  const shown = path.startsWith(`${home}/`) ? `~${path.slice(home.length)}` : path;
  return truncate(shown, 60);
}

function argumentOf(call: ToolCall): string {
  const input = call.input ?? {};
  const firstString = Object.values(input).find((value) => typeof value === "string");
  switch (call.name) {
    case "Bash": return truncate(String(input.command ?? ""), 60);
    case "Read": {
      const range = call.readRange;
      const path = displayPath(String(input.file_path ?? ""));
      return range ? `${path} L${range.startLine}-${range.endLine}` : path;
    }
    case "Edit":
    case "Write": return displayPath(String(input.file_path ?? ""));
    case "Glob":
    case "Grep": return truncate(String(input.pattern ?? ""), 60);
    case "Agent": return truncate(String(input.description ?? ""), 60);
    case "Skill": return truncate(String(input.skill ?? input.name ?? ""), 60);
    default: return firstString === undefined ? "" : truncate(firstString, 40);
  }
}

export function toolLine(call: ToolCall): string {
  const glyph = call.status === "complete" ? "✓" : "✗";
  const head = [glyph, call.name, argumentOf(call)].filter((part) => part !== "").join(" ");
  const tail: string[] = [];
  if (call.status === "timed_out") tail.push("timed out");
  else if (call.status === "error") tail.push(truncate(firstLine(call.output), 80));
  else if (call.name === "Bash" && call.exitCode !== undefined && call.exitCode !== null) {
    tail.push(`exit ${call.exitCode}`);
  }
  if (call.durationMs !== undefined && call.status !== "running" && call.status !== "queued") {
    tail.push(`${(call.durationMs / 1_000).toFixed(1)}s`);
  }
  return tail.length === 0 ? head : `${head} · ${tail.join(" · ")}`;
}

export function toolStatus(call: ToolCall): string {
  return [call.name, argumentOf(call)].filter((part) => part !== "").join(" ").slice(0, 60);
}

function firstLine(output: string): string {
  return output.split("\n", 1)[0] ?? "";
}
```

```ts
import type { Message, Session, SessionSummary } from "../types.js";

async function responseError(response: Response): Promise<string> {
  try {
    return ((await response.json()) as { error?: string }).error ?? `Request failed (${response.status})`;
  } catch {
    return `Request failed (${response.status})`;
  }
}

export class AmberApi {
  constructor(private readonly baseUrl: string) {}

  private url(path: string): string { return `${this.baseUrl}${path}`; }

  private async request(path: string, init?: RequestInit): Promise<unknown> {
    const response = await fetch(this.url(path), {
      ...init,
      headers: { "content-type": "application/json", ...init?.headers },
    });
    if (!response.ok) throw new Error(await responseError(response));
    return response.json().catch(() => undefined);
  }

  async listSessions(): Promise<SessionSummary[]> {
    const body = await this.request("/api/sessions") as { sessions?: SessionSummary[] };
    return body.sessions ?? [];
  }

  async createSession(path: string, name?: string): Promise<Session> {
    const body = await this.request("/api/sessions", {
      method: "POST",
      body: JSON.stringify({ path, ...(name ? { name } : {}) }),
    }) as { session: Session };
    return body.session;
  }

  async history(sessionId: string, before?: string): Promise<{ messages: Message[]; hasMore: boolean }> {
    const query = before === undefined ? "" : `?before=${encodeURIComponent(before)}`;
    return await this.request(`/api/sessions/${sessionId}/messages${query}`) as {
      messages: Message[];
      hasMore: boolean;
    };
  }

  async sendMessage(sessionId: string, content: string): Promise<void> {
    // The run's events arrive on the /events subscription; drain this body.
    const response = await fetch(this.url(`/api/sessions/${sessionId}/messages`), {
      method: "POST",
      headers: { "content-type": "application/json" },
      body: JSON.stringify({ content }),
    });
    if (!response.ok) throw new Error(await responseError(response));
    await response.body?.cancel();
  }

  async queuedMessage(sessionId: string, content: string, kind: "message" | "command"): Promise<"queued" | "busy"> {
    const response = await fetch(this.url(`/api/sessions/${sessionId}/queued-message`), {
      method: "POST",
      headers: { "content-type": "application/json" },
      body: JSON.stringify({ content, kind }),
    });
    if (response.status === 409) return "busy";
    if (!response.ok) throw new Error(await responseError(response));
    return "queued";
  }

  async command(sessionId: string, command: string): Promise<Record<string, unknown>> {
    return await this.request(`/api/sessions/${sessionId}/commands`, {
      method: "POST",
      body: JSON.stringify({ command }),
    }) as Record<string, unknown>;
  }

  async abort(sessionId: string): Promise<void> {
    await this.request(`/api/sessions/${sessionId}/abort`, { method: "POST" });
  }

  async answerQuestions(
    sessionId: string,
    toolUseId: string,
    answers: Record<string, string> | { cancelled: true },
  ): Promise<void> {
    await this.request(`/api/sessions/${sessionId}/questions/${encodeURIComponent(toolUseId)}/answers`, {
      method: "POST",
      body: JSON.stringify(answers),
    });
  }

  async planDecision(
    sessionId: string,
    toolUseId: string,
```

```ts
import { readServerSentEvents } from "../sse.js";

export interface CliSessionEvent { event: string; data: unknown }

const DEFAULT_BACKOFF_MS = [1_000, 2_000, 5_000, 10_000];

/** Subscribes to /api/sessions/:id/events forever, reconnecting with capped
 * backoff. Every (re)connection starts with a snapshot the caller diffs. */
export async function subscribeSessionEvents(
  baseUrl: string,
  sessionId: string,
  onEvent: (event: CliSessionEvent) => void,
  signal: AbortSignal,
  backoffMs: number[] = DEFAULT_BACKOFF_MS,
): Promise<void> {
  let attempt = 0;
  while (!signal.aborted) {
    try {
      const response = await fetch(`${baseUrl}/api/sessions/${sessionId}/events`, { signal });
      if (!response.ok || !response.body) throw new Error(`event stream failed (${response.status})`);
      attempt = 0;
      for await (const { event, data } of readServerSentEvents(response)) {
        onEvent({ event, data: JSON.parse(data) });
      }
      throw new Error("event stream ended");
    } catch {
      if (signal.aborted) return;
      const delay = backoffMs[Math.min(attempt, backoffMs.length - 1)]!;
      attempt += 1;
      await new Promise<void>((resolve) => {
        const timer = setTimeout(resolve, delay);
        signal.addEventListener("abort", () => { clearTimeout(timer); resolve(); }, { once: true });
      });
    }
  }
}
```

```ts
export function parseCliArgs(argv: readonly string[], env: NodeJS.ProcessEnv = process.env): ParsedCliArgs {
  const args: CliArgs = {
    server: env.AMBER_SERVER && env.AMBER_SERVER !== "" ? env.AMBER_SERVER : "http://127.0.0.1:3000",
    history: 2,
    list: false,
  };
  const positional: string[] = [];
  for (let i = 0; i < argv.length; i += 1) {
    const argument = argv[i]!;
    if (argument === "--help" || argument === "-h") return { kind: "help" };
    if (argument === "--server") {
      const value = argv[i + 1];
      if (!value) throw new Error("--server requires a url");
      args.server = value;
      i += 1;
    } else if (argument === "--history") {
      const value = Number(argv[i + 1]);
      if (!Number.isInteger(value) || value < 0) throw new Error("--history requires a non-negative integer");
      args.history = value;
      i += 1;
    } else if (argument === "--list") {
      args.list = true;
    } else if (argument === "--new") {
      const value = argv[i + 1];
      if (value !== undefined && !value.startsWith("--")) args.newName = value;
      else args.newName = "";
      if (args.newName !== "") i += 1;
    } else if (argument.startsWith("--")) {
      throw new Error(`unknown option: ${argument}`);
    } else {
      positional.push(argument);
    }
  }
  if (positional.length > 1) throw new Error("only one session id may be given");
  if (positional.length === 1) args.session = positional[0];
  return { kind: "args", args };
}
```

```ts
  answerQuestions(sessionId: string, toolUseId: string, answers: Record<string, string> | { cancelled: true }): Promise<void>;
  planDecision(sessionId: string, toolUseId: string, decision: PlanDecision): Promise<void>;

  question: string;
  header?: string;
  options: Array<{ label: string; description?: string }>;
  multiSelect?: boolean;

const SKIPPED_KINDS = new Set(["tool-result", "skill", "skill-catalog", "agent-notification"]);

function truncate(text: string, max: number): string {
  const flat = text.replace(/\s+/g, " ").trim();
  return flat.length > max ? `${flat.slice(0, max - 1)}…` : flat;
}

function firstLine(text: string): string {
  return text.split("\n", 1)[0] ?? "";
}

export class SessionDisplay {
  active = false;
  gatePending = false;
  private printedBlocks = new Map<string, number>();
  private printedUserIds = new Set<string>();
  private printedToolIds = new Set<string>();
  private printedOrder: string[] = [];
  private splitter: BlockSplitter | null = null;
  private streamingMessageId: string | null = null;
  private streamingChars = 0;
  private thinkingChars = 0;
  private lastCompletedTool: ToolCall | null = null;

    private readonly hooks: DisplayHooks,

  attach(messages: Message[], limit: number): void {

    for (const message of limit > 0 ? rendered.slice(-limit) : []) this.printFromHistory(message);
  }

  latestToolOutput(): string | null {
    return this.lastCompletedTool?.output ?? null;
  }

  oldestPrintedId(): string | undefined {
    return this.printedOrder[0];
  }

  async applyEvent(event: string, data: unknown): Promise<void> {
    const payload = data as Record<string, any>;
    switch (event) {
      case "snapshot":
        this.setActive(payload.session?.active === true);
        for (const message of payload.session?.messages ?? []) {
          if (SKIPPED_KINDS.has(message.kind ?? "")) continue;
          this.printFromHistory(message);

      case "start":

        // The terminal already echoed this input; only mark it seen so a
        // later snapshot does not reprint it.

          this.markPrinted(payload.userMessage.id);
        }

      case "continuation":
        this.beginStream(payload.assistantMessage.id);
        break;
      case "delta": {
        const text = String(payload.text ?? "");
        this.ensureSplitter();
        for (const block of this.splitter!.push(text)) this.printStreamBlock(block);
        this.streamingChars += text.length;
        this.paintStatus();
        break;

      case "thinking_delta":
        this.thinkingChars += String(payload.thinking ?? "").length;
        this.paintStatus();

      case "tool_update":
        this.applyToolUpdate(payload.toolCall as ToolCall);
        break;
      case "tool_output":
        break; // live tool output is never printed
      case "user_message":
        this.printUser(payload.message as Message);
        break;
      case "assistant_complete":
        this.finishStream(payload.message as Message);
        break;
      case "done":
        this.finishStream(payload.message as Message);
        this.printUsage(payload.message as Message);
        this.setActive(false);
        break;
      case "error":
        this.splitter = null;
        this.term.line(this.term.style(`✗ error: ${truncate(String(payload.error ?? ""), 160)}`, RED));
        this.term.flush();
        this.setActive(false);
        break;
      case "session_named":
        this.term.line(this.term.style(`· named: ${truncate(String(payload.title ?? ""), 60)}`, DIM));
        break;
      case "plan_mode_state":
        this.term.line(this.term.style(`· plan mode ${payload.planMode?.active ? "on" : "off"}`, DIM));
        break;
      case "plan_mode_request":
        await this.handlePlanModeRequest(payload);
        break;
      case "ask_user_question":
        await this.handleQuestions(payload);
        break;
      case "compaction_start":
        this.setActive(true);
        this.term.status("compacting…");
        break;
      case "compaction_progress":
        this.term.status(`compacting… ${payload.generatedCharacters ?? 0} chars`);
        break;
      case "compaction_complete":
        this.term.line(this.term.style("· compacted", DIM));
        this.setActive(false);
        break;
      case "compaction_error":
        this.term.line(this.term.style(`✗ compaction failed: ${truncate(String(payload.error ?? ""), 120)}`, RED));
        this.setActive(false);
        break;
      default:
        break;
    }
  }

  private setActive(active: boolean): void {
    if (this.active === active) return;
    this.active = active;
    if (!active) this.term.status(null);
    this.hooks.onActiveChange(active);
  }

  private beginStream(messageId: string): void {
    this.streamingMessageId = messageId;
    this.splitter = null;
    this.streamingChars = 0;
    this.thinkingChars = 0;
    if (!this.printedBlocks.has(messageId)) this.printedBlocks.set(messageId, 0);
    this.setActive(true);
  }

  private ensureSplitter(): void {
    if (this.splitter === null) {
      this.splitter = createBlockSplitter();
      this.streamingChars = 0;
    }
  }

  private printStreamBlock(block: string): void {
    const rendered = renderMarkdown(block, { width: this.term.columns, color: this.term.color });
    this.term.line("");
    if (rendered !== "") this.term.line(rendered);
    const id = this.streamingMessageId ?? "stream";
    this.printedBlocks.set(id, (this.printedBlocks.get(id) ?? 0) + 1);
  }

  private finishStream(message: Message): void {
    if (this.splitter !== null && message.id === this.streamingMessageId) {
      const remainder = this.splitter.finish();
      if (remainder !== "") this.printStreamBlock(remainder);
    }
    this.splitter = null;
    if (!this.printedBlocks.has(message.id)) this.printedBlocks.set(message.id, blocksOf(message.content));
    if (!this.printedOrder.includes(message.id)) this.printedOrder.push(message.id);
  }

  private printUsage(message: Message): void {
    if (!message.usage) return;
    this.term.line("");
    this.term.line(
      this.term.style(`· ${formatTokenCountInThousands(message.usage.input)}→${formatTokenCountInThousands(message.usage.output)} tok`, DIM),
    );
  }

  private applyToolUpdate(call: ToolCall): void {
    if (call.status === "running" || call.status === "queued") {
      if (call.status === "running") this.paintStatus(call);
      return;
    }
    if (this.printedToolIds.has(call.id)) return;
    this.printedToolIds.add(call.id);
    this.lastCompletedTool = call;
    const codes = call.status === "complete" ? DIM : RED;
    this.term.line(this.term.style(toolLine(call), codes));
  }

  private paintStatus(runningTool?: ToolCall): void {
    if (!this.active) return;
    const parts: string[] = [];
    if (this.thinkingChars > 0) parts.push(`thinking ${this.thinkingChars}`);
    if (this.streamingChars > 0) parts.push(`streaming ${this.streamingChars}`);
    if (runningTool) parts.push(`${toolStatus(runningTool)} …`);
    this.term.status(parts.length > 0 ? parts.join(" · ") : "working…");
  }

  private printUser(message: Message): void {
    if (this.printedUserIds.has(message.id)) return;
    this.printedUserIds.add(message.id);
    this.markPrinted(message.id);
    this.term.line("");
    this.term.line(this.term.style(`> ${truncate(message.content, this.term.columns - 4)}`, DIM));
  }

  /** History/snapshot path: prints blocks of `message` beyond what was printed. */
  private printFromHistory(message: Message): void {
    if (message.role === "user") {
      this.printUser(message);
      return;
    }
    if (message.kind !== undefined && message.kind !== "chat") {
      if (!this.printedBlocks.has(message.id)) {
        this.markPrinted(message.id);
        this.printedBlocks.set(message.id, blocksOf(message.content));
        this.term.line("");
        this.term.line(this.term.style(`· ${truncate(firstLine(message.content), this.term.columns - 4)}`, DIM));
      }
      return;
    }
    const seen = this.printedBlocks.get(message.id) ?? 0;
    const blocks = closedBlocksOf(message.content);
    for (let index = seen; index < blocks.length; index += 1) {
      this.term.line("");
      const rendered = renderMarkdown(blocks[index]!, { width: this.term.columns, color: this.term.color });
      if (rendered !== "") this.term.line(rendered);
    }
    if (!this.printedOrder.includes(message.id)) this.markPrinted(message.id);
    this.printedBlocks.set(message.id, blocks.length);
  }

  private markPrinted(id: string): void {
    if (!this.printedOrder.includes(id)) this.printedOrder.push(id);
  }

  private async handleQuestions(payload: Record<string, any>): Promise<void> {
    this.term.flush();
    this.gatePending = true;
    try {
      const answers: Record<string, string> = {};
      for (const question of (payload.questions ?? []) as GateQuestion[]) {
        this.term.line("");
        this.term.line(question.header ? `${question.header}: ${question.question}` : question.question);
        question.options.forEach((option, index) => {
          const description = option.description ? ` — ${truncate(option.description, 60)}` : "";
          this.term.line(`  ${index + 1}) ${option.label}${description}`);
        });
        if (question.multiSelect) this.term.line("  (comma-separated numbers)");
        const input = await this.hooks.ask("answer: ");
        answers[question.question] = this.parseAnswer(input, question);
      }
      const declined = Object.values(answers).every((answer) => answer === "");
      await this.hooks.answerQuestions(
        this.sessionId,
        String(payload.toolUseId),
        declined ? { cancelled: true } : answers,
      );
    } finally {
      this.gatePending = false;
      this.term.flush();
    }
  }

  private parseAnswer(input: string, question: GateQuestion): string {
    const trimmed = input.trim();
    if (trimmed === "") return "";
    const labels = trimmed
      .split(/[,\s]+/)
      .filter(Boolean)
      .map((pick) => (/^\d+$/.test(pick) ? question.options[Number(pick) - 1]?.label : undefined))
      .filter((label): label is string => label !== undefined);
    return labels.length === 0 ? trimmed : labels.join(", ");
  }

  private async handlePlanModeRequest(payload: Record<string, any>): Promise<void> {
    this.term.flush();
    this.gatePending = true;
    try {
      let decision: PlanDecision;
      if (payload.kind === "enter") {
        const answer = (await this.hooks.ask("enter plan mode? [y/N] ")).trim().toLowerCase();
        decision = answer === "y" || answer === "yes" ? { approved: true } : { approved: false };
      } else {
        this.term.line("");
        const plan = renderMarkdown(String(payload.plan ?? ""), { width: this.term.columns, color: this.term.color });
        if (plan !== "") this.term.line(plan);
        const answer = (await this.hooks.ask("approve plan? [y]es / [n]o / [s]eparate session ")).trim().toLowerCase();
        decision = answer.startsWith("s") ? { newSession: true } : answer.startsWith("y") ? { approved: true } : { approved: false };
      }
      await this.hooks.planDecision(this.sessionId, String(payload.toolUseId), decision);
    } finally {
      this.gatePending = false;
      this.term.flush();
    }
  }
}

/** Closed blocks plus the remainder — the deterministic split live streaming uses. */
function closedBlocksOf(content: string): string[] {
  const splitter = createBlockSplitter();
  const blocks = splitter.push(content);
  const remainder = splitter.finish();
  return remainder === "" ? blocks : [...blocks, remainder];
}

function blocksOf(content: string): number {
  return closedBlocksOf(content).length;
}
```

```ts
export async function runCli(argv: readonly string[]): Promise<number> {
  let parsed: ParsedCliArgs;
  try {
    parsed = parseCliArgs(argv);
  } catch (error) {
    process.stderr.write(`amber cli: ${error instanceof Error ? error.message : String(error)}\n\n${cliUsageText()}\n`);
    return 2;
  }
  if (parsed.kind === "help") {
    process.stdout.write(`${cliUsageText()}\n`);
    return 0;
  }
  const args = parsed.args;
  const term = new Terminal();
  const api = new AmberApi(args.server);
  try {
    if (args.list) return await listSessions(api, term);
    const sessionId = await resolveSessionId(api, args);
    term.line(term.style(`· session ${sessionId.slice(0, 8)}`, DIM));
    return await repl(api, term, sessionId, args);
  } catch (error) {
    term.line(term.style(`✗ ${error instanceof Error ? error.message : String(error)}`, RED));
    term.end();
    return 1;
  }
}

async function listSessions(api: AmberApi, term: Terminal): Promise<number> {
  for (const session of await api.listSessions()) {
    term.line(`${session.id.slice(0, 8)}  ${session.updatedAt.slice(0, 10)}  ${session.title}`);
  }
  term.end();
  return 0;
}

async function resolveSessionId(api: AmberApi, args: CliArgs): Promise<string> {
  if (args.newName !== undefined) {
    const session = await api.createSession(process.cwd(), args.newName === "" ? undefined : args.newName);
    return session.id;
  }
  const sessions = await api.listSessions();
  if (args.session !== undefined) {
    const matches = sessions.filter((session) => session.id.startsWith(args.session!));
    if (matches.length === 0) throw new Error(`no session matches '${args.session}'`);
    if (matches.length > 1) throw new Error(`session prefix '${args.session}' is ambiguous`);
    return matches[0]!.id;
  }
  if (sessions.length === 0) {
    const session = await api.createSession(process.cwd());
    return session.id;
  }
  return sessions[0]!.id;
}

async function repl(api: AmberApi, term: Terminal, sessionId: string, args: CliArgs): Promise<number> {
  const rl = createInterface({ input: process.stdin, output: process.stdout, prompt: "> " });
  let currentSessionId = sessionId;
  let subscription = new AbortController();
  let display = createDisplay(currentSessionId);

  function createDisplay(id: string): SessionDisplay {
    return new SessionDisplay(term, id, {
      ask: async (prompt) => await rl.question(prompt),
      answerQuestions: (sid, toolUseId, answers) => api.answerQuestions(sid, toolUseId, answers),
      planDecision: (sid, toolUseId, decision) => api.planDecision(sid, toolUseId, decision),
      onActiveChange: (active) => {
        rl.setPrompt(active ? "" : "> ");
        if (!active) {
          term.flush();
          rl.prompt();
        }
      },
    });
  }

  const dispatch = (event: string, data: unknown) => {
    void display.applyEvent(event, data).catch((error) => {
      term.line(term.style(`✗ ${error instanceof Error ? error.message : String(error)}`, RED));
    });
  };
  void subscribeSessionEvents(args.server, currentSessionId, dispatch, subscription.signal);

  const history = await api.history(currentSessionId);
  display.attach(history.messages, args.history);
  term.flush();

  let lastSigint = 0;
  const exit = (code: number): never => {
    term.end();
    subscription.abort();
    rl.close();
    process.exit(code);
  };
  rl.on("SIGINT", () => {
    const now = Date.now();
    if (now - lastSigint < 1_000) exit(0);
    lastSigint = now;
    if (display.active) {
      void api.abort(currentSessionId).catch(() => undefined);
      term.status(null);
      term.flush();
    } else {
      exit(0);
    }
  });
  process.once("SIGINT", () => exit(130));
  rl.on("close", () => exit(0));
  if (process.stdin.isTTY) process.stdin.on("data", () => term.suppressStatus());

  rl.on("line", (input) => {
    if (display.gatePending) return; // consumed by a gate's rl.question
    const text = input.trim();
    if (text === "") {
      if (!display.active) rl.prompt();
      return;
    }
    void handleInput(text)
      .catch((error) => {
        term.line(term.style(`✗ ${error instanceof Error ? error.message : String(error)}`, RED));
      })
      .finally(() => {
        term.flush();
        if (!display.active) rl.prompt();
      });
  });

  async function handleInput(text: string): Promise<void> {
    if (text === "/quit") exit(0);
    if (text === "/list") {
      for (const session of await api.listSessions()) {
        term.line(`${session.id.slice(0, 8)}  ${session.updatedAt.slice(0, 10)}  ${session.title}`);
      }
      return;
    }
    if (text === "/output" || text.startsWith("/output ")) {
      const chars = numberArgument(text, 2_000);
      const output = display.latestToolOutput();
      term.line(output === null ? term.style("· no completed tool call yet", DIM) : output.slice(-chars));
      return;
    }
    if (text === "/history" || text.startsWith("/history ")) {
      const count = numberArgument(text, 5);
      const page = await api.history(currentSessionId, display.oldestPrintedId());
      display.attach(page.messages, count);
      return;
    }
    if (text === "/new" || text.startsWith("/new ")) {
      const name = text.split(/\s+/).slice(1).join(" ");
      const session: Session = await api.createSession(process.cwd(), name === "" ? undefined : name);
      subscription.abort();
      currentSessionId = session.id;
      subscription = new AbortController();
      display = createDisplay(currentSessionId);
      void subscribeSessionEvents(args.server, currentSessionId, dispatch, subscription.signal);
      term.line(term.style(`· session ${session.id.slice(0, 8)}`, DIM));
      return;
    }
    if (text.startsWith("/")) {
      if (display.active) await queue(api, term, currentSessionId, text, "command");
      else {
        await api.command(currentSessionId, text);
        term.line(term.style("· done", DIM));
      }
      return;
    }
    if (display.active) await queue(api, term, currentSessionId, text, "message");
    else await api.sendMessage(currentSessionId, text);
  }

  rl.prompt();
  return await new Promise<number>(() => undefined); // exits via exit()
}

async function queue(
  api: AmberApi,
  term: Terminal,
  sessionId: string,
  content: string,
  kind: "message" | "command",
): Promise<void> {
  const result = await api.queuedMessage(sessionId, content, kind);
  if (result === "queued") term.line(term.style("· queued", DIM));
  else term.line(term.style("· run ended before queueing; send again", DIM));
}

function numberArgument(text: string, fallback: number): number {
  const value = Number(text.split(/\s+/)[1]);
  return Number.isInteger(value) && value > 0 ? value : fallback;
}
```
